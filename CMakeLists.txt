cmake_minimum_required(VERSION 3.22)
project(duckmidi VERSION 0.10.0 LANGUAGES C CXX)
set(DUCKMIDI_DELIVERY_VERSION "${PROJECT_VERSION}" CACHE STRING "Application and delivery version")
if(NOT DUCKMIDI_DELIVERY_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "DUCKMIDI_DELIVERY_VERSION must contain three numeric parts, such as 0.10.0.")
endif()
add_compile_definitions(DM_APP_VERSION="${DUCKMIDI_DELIVERY_VERSION}")
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(WIN32)
  list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL3-3.2.28/x86_64-w64-mingw32")
endif()
find_package(SDL3 3.2 CONFIG REQUIRED)
if(WIN32)
  add_library(fluidsynth SHARED IMPORTED)
  set_target_properties(fluidsynth PROPERTIES IMPORTED_IMPLIB "${CMAKE_CURRENT_SOURCE_DIR}/vendor/fluidsynth-2.5.4/fluidsynth-v2.5.4-win10-x64-glib/lib/libfluidsynth-3.lib" IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/vendor/fluidsynth-2.5.4/fluidsynth-v2.5.4-win10-x64-glib/bin/libfluidsynth-3.dll" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/fluidsynth-2.5.4/fluidsynth-v2.5.4-win10-x64-glib/include")
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FLUID REQUIRED IMPORTED_TARGET fluidsynth>=2.5.4)
  add_library(fluidsynth ALIAS PkgConfig::FLUID)
endif()
add_library(duckcore STATIC shared/spectrum.c src/core.c src/midi.c src/project.c src/instruments.c)
add_subdirectory(libraries/m64export)
include(tools/ffmpeg.cmake)
add_library(duckmedia STATIC shared/audio_export.c)
target_link_libraries(duckmedia PUBLIC duckcore ${FFMPEG_LIBRARIES})
target_include_directories(duckcore PUBLIC src)
if(NOT WIN32)
  target_link_libraries(duckcore PUBLIC m)
endif()
set(IMGUI vendor/imgui-1.92.5)
set(RTMIDI vendor/rtmidi-6.0.0)
# Compile assets with the same C++ toolchain as the editor; no shell/interpreter.
add_executable(duck_icon_baker tools/bake-icons.cpp)
if(MINGW)
  target_link_options(duck_icon_baker PRIVATE -municode -static-libgcc -static-libstdc++)
endif()
file(GLOB ICON_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/iconoir/*.svg")
set(ICON_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${ICON_OUTPUT}/toolbar_icons.h" "${ICON_OUTPUT}/piano_icon.h"
  COMMAND duck_icon_baker "${CMAKE_CURRENT_SOURCE_DIR}" "${ICON_OUTPUT}"
  DEPENDS duck_icon_baker assets/piano.svg assets/piano-outline.svg assets/cut-copy.svg ${ICON_SOURCES}
  VERBATIM)
add_custom_target(duck_icons DEPENDS "${ICON_OUTPUT}/toolbar_icons.h" "${ICON_OUTPUT}/piano_icon.h")
add_library(duckgui STATIC src/ui.cpp ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_sdl3.cpp ${IMGUI}/backends/imgui_impl_sdlrenderer3.cpp)
add_dependencies(duckgui duck_icons)
target_include_directories(duckgui PRIVATE src ${IMGUI} ${IMGUI}/backends "${ICON_OUTPUT}")
target_link_libraries(duckgui PUBLIC SDL3::SDL3)
add_library(duckmidi_io STATIC ${RTMIDI}/RtMidi.cpp ${RTMIDI}/rtmidi_c.cpp)
target_include_directories(duckmidi_io PUBLIC ${RTMIDI})
if(WIN32)
  target_compile_definitions(duckmidi_io PRIVATE __WINDOWS_MM__)
  target_link_libraries(duckmidi_io PRIVATE winmm ole32 uuid)
elseif(APPLE)
  target_compile_definitions(duckmidi_io PRIVATE __MACOSX_CORE__)
  target_link_libraries(duckmidi_io PRIVATE "-framework CoreMIDI" "-framework CoreAudio" "-framework CoreFoundation")
else()
  find_package(ALSA REQUIRED)
  find_package(Threads REQUIRED)
  target_compile_definitions(duckmidi_io PRIVATE __LINUX_ALSA__)
  target_link_libraries(duckmidi_io PRIVATE ALSA::ALSA Threads::Threads)
endif()
add_executable(duckmidi src/main.c src/editor.c src/editor_browser.c src/editor_dialogs.c src/editor_event_view.c src/editor_events.c src/editor_input.c src/editor_midi_inspector.c src/editor_notes.c src/editor_pitch.c src/editor_playlist.c src/editor_preferences.c src/editor_project.c src/editor_selection.c src/editor_shell.c src/editor_song_events.c src/editor_state.c src/editor_timeline.c src/editor_transport.c src/audio.c src/preferences.c)
target_sources(duckmidi PRIVATE shared/pattern_generation.c src/editor_generate.c src/editor_percussion.c src/editor_note_ports.c src/editor_save.c src/export_plugins.c src/editor_export.c src/editor_export_ui.c)
add_dependencies(duckmidi m64export)
add_custom_command(TARGET duckmidi POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:duckmidi>/plugins"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:m64export>" "$<TARGET_FILE_DIR:duckmidi>/plugins/$<TARGET_FILE_NAME:m64export>")
target_link_libraries(duckmidi PRIVATE duckcore duckgui duckmidi_io fluidsynth duckmedia)
set_target_properties(duckmidi PROPERTIES LINKER_LANGUAGE CXX)
set_target_properties(duckmidi PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
if(MINGW)
  target_link_options(duckmidi PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(m64export PRIVATE -static-libgcc)
endif()
if(WIN32)
  set_target_properties(duckmidi PROPERTIES WIN32_EXECUTABLE TRUE)
  enable_language(RC)
  target_sources(duckmidi PRIVATE tools/app.rc)
  target_include_directories(duckmidi PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(WIN32)
  add_custom_command(TARGET duckmidi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FFMPEG_DLLS} "$<TARGET_FILE_DIR:duckmidi>" COMMAND_EXPAND_LISTS)
  file(GLOB FLUID_DLLS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/fluidsynth-2.5.4/fluidsynth-v2.5.4-win10-x64-glib/bin/*.dll")
  foreach(DLL IN LISTS FLUID_DLLS)
    if(NOT DLL MATCHES "SDL3.dll$")
      add_custom_command(TARGET duckmidi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DLL}" "$<TARGET_FILE_DIR:duckmidi>")
    endif()
  endforeach()
  add_custom_command(TARGET duckmidi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL3::SDL3>" "$<TARGET_FILE_DIR:duckmidi>")
  get_filename_component(COMPILER_BIN "${CMAKE_C_COMPILER}" DIRECTORY)
  if(EXISTS "${COMPILER_BIN}/libwinpthread-1.dll")
    add_custom_command(TARGET duckmidi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${COMPILER_BIN}/libwinpthread-1.dll" "$<TARGET_FILE_DIR:duckmidi>")
  endif()
endif()
include(tools/package.cmake)
include(CTest)

  target_link_libraries(audio_export_test PRIVATE duckmedia)

  file(MAKE_DIRECTORY "${PROJECT_SOURCE_DIR}/agents/reports/audio-export")
  add_test(NAME audio_export_formats COMMAND audio_export_test "${PROJECT_SOURCE_DIR}/agents/reports/audio-export")
  add_executable(m64export_api_test tests/m64export_api_test.c)
  target_link_libraries(m64export_api_test PRIVATE m64export_static)
  add_test(NAME native_m64_export_api COMMAND m64export_api_test)
  add_executable(core_test tests/core_test.c)
  target_link_libraries(core_test PRIVATE duckcore)
  add_test(NAME midi_project_core COMMAND core_test)
  add_executable(spectrum_test tests/spectrum_test.c)
  target_link_libraries(spectrum_test PRIVATE duckcore)
  add_test(NAME spectrum_analysis COMMAND spectrum_test)
  add_executable(ports_test tests/ports_test.c)
  target_link_libraries(ports_test PRIVATE duckcore)
  add_test(NAME midi_ports COMMAND ports_test)
  set_tests_properties(midi_ports PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" LABELS ports)
  add_executable(release_export_test tests/release_export_test.c src/export_plugins.c)
  target_include_directories(release_export_test PRIVATE src)
  target_link_libraries(release_export_test PRIVATE SDL3::SDL3)
  add_dependencies(release_export_test duckmidi)
  set_tests_properties(midi_project_core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  foreach(TEST_TARGET audio_export_test m64export_api_test core_test spectrum_test ports_test release_export_test)
    if(MSVC)
      target_compile_options(${TEST_TARGET} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${TEST_TARGET} PRIVATE -UNDEBUG)
    endif()
  endforeach()
  option(DUCKMIDI_DESKTOP_TESTS "Build the Windows desktop/audio regression suites" OFF)
  if(DUCKMIDI_DESKTOP_TESTS)
    add_subdirectory(tests)
  endif()
endif()
